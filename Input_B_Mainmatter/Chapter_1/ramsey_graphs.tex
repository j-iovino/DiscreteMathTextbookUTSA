% --- Color choosers for Figs 1–4 Grid points
\newcommand{\randcoin}{
  \pgfmathparse{rnd<0.5?"red":"blue"}\edef\col{\pgfmathresult}%
}
\newcommand{\allred}{\def\col{red}}
\newcommand{\allblue}{\def\col{blue}}
\newcommand{\randSpectrum}{%
  \pgfmathtruncatemacro{\p}{round(100*rnd)}%
  \edef\col{red!\p!blue}%
}

% --- Figure 5: LIGHT at (n1,n1) -> DEEP at (n25,n25)  (comment only; 25x25 uses N-version below)
\newcommand{\originToTipPurple}{%
  \pgfmathsetmacro{\s}{((\i-1)+(\j-1))/18} % 0 at (1,1), 1 at (10,10)
  \pgfmathsetmacro{\t}{pow(\s,1.2)}        % optional contrast; tweak 1.0–1.6
  \pgfmathtruncatemacro{\p}{round(100*(1-\t))}
  \edef\col{LP!\p!DP}%
}

% --- Standalone purple arrow legend (center of Page 2 row)
\newcommand{\ArrowLegendStandalone}{%
  \begin{tikzpicture}[scale=1]
    \def\W{10}   % arrow width (tail at -\W/2, tip at +\W/2)
    \def\H{1.2}  % arrow nominal height
    \def\S{220}  % number of strips (higher = smoother)

    \begin{scope}
      % Arrow clip
      \path[clip]
        (-\W/2,0) -- ++(\W*0.72,0) -- ++(0,-\H*0.18) --
        ++(\W*0.28,\H*0.50) -- ++(-\W*0.28,\H*0.50) --
        ++(0,-\H*0.18) -- ++(-\W*0.72,0) -- cycle;

      % Progressive DP→LP (as in your current file)
      \pgfmathsetmacro{\w}{\W/\S}
      \foreach \k in {0,...,\S}{
        \pgfmathsetmacro{\xnorm}{\k/\S}
        \pgfmathsetmacro{\gamma}{pow(\xnorm,1.2)}        % progression
        \pgfmathtruncatemacro{\p}{round(100*(1-\gamma))} % reverse order
        \path[fill=LP!\p!DP,draw=none]
          (-\W/2+\k*\w,-2*\H) rectangle ++(\w,4*\H);
      }
    \end{scope}

    % Outline and labels
    \draw
      (-\W/2,0) -- ++(\W*0.72,0) -- ++(0,-\H*0.18) --
      ++(\W*0.28,\H*0.50) -- ++(-\W*0.28,\H*0.50) --
      ++(0,-\H*0.18) -- ++(-\W*0.72,0) -- cycle;

    \node[below] at (-\W/2,-\H/2) {Light};
    \node[below] at ( \W/2,-\H/2) {Deep};
    \node[above,yshift=8pt] at (0,\H/2) {Transition};
  \end{tikzpicture}
}

% Grid size controller for general N (we'll set N=25)
\newcommand{\setGridN}[1]{%
  \def\N{#1}%
  \pgfmathsetmacro{\Den}{2*(#1-1)}% for (i-1+j-1) normalization up to 2*(N-1)
}

% Purple gradient normalized to \N (Light -> Deep along origin->tip)
\newcommand{\originToTipPurpleN}{%
  \pgfmathsetmacro{\s}{min(1,max(0,((\i-1)+(\j-1))/\Den))} % clamp to [0,1]
  \pgfmathsetmacro{\t}{pow(\s,1.2)}          % contrast tweak
  \pgfmathtruncatemacro{\p}{round(100*(1-\t))}
  \edef\col{LP!\p!DP}%
}


% Red→Blue horizontal bar legend (used INSIDE Figure 4)
\newcommand{\ColorbarRB}[5]{% (#1,#2)=LL, width=#3, height=#4
  \begin{scope}
    \pgfmathsetmacro{\w}{#3/100}
    \foreach \k in {0,...,100}{
    \pgfmathtruncatemacro{\p}{\k}
  % Blue (left) -> Red (right), matching labels
  \path[fill=blue!\p!red,draw=none]
    (#1+\k*\w,#2) rectangle ++(\w,#4);
}

    \draw (#1,#2) rectangle ++(#3,#4);
    % Leave your labels as-is (they’ll now match the colors):
    % Left label:
    \node[below] at (#1,#2) {Blue = 0};
    % Right label:
    \node[below] at (#1+#3,#2) {Red = 1};
  \end{scope}
}

% Legend: Red (left) -> Blue (right), labels: Red=0 (left), Blue=1 (right)
\newcommand{\LegendRB}[4]{% (#1,#2)=LL, width=#3, height=#4
  \begin{scope}
    \pgfmathsetmacro{\w}{#3/100}
    \foreach \k in {0,...,100}{
      \pgfmathtruncatemacro{\p}{\k}
      \path[fill=red!\p!blue,draw=none]
        (#1+\k*\w,#2) rectangle ++(\w,#4);
    }
    \draw (#1,#2) rectangle ++(#3,#4);
    \node[below] at (#1,#2) {Red = 0};
    \node[below] at (#1+#3,#2) {Blue = 1};
  \end{scope}
}

% Legend: Blue (left) -> Red (right), labels: Blue=0 (left), Red=1 (right)
\newcommand{\ColorbarBR}[4]{% (#1,#2)=LL, width=#3, height=#4
  \begin{scope}
    \pgfmathsetmacro{\w}{#3/100}
    \foreach \k in {0,...,100}{
      \pgfmathtruncatemacro{\p}{\k}
      \path[fill=blue!\p!red,draw=none]
        (#1+\k*\w,#2) rectangle ++(\w,#4);
    }
    \draw (#1,#2) rectangle ++(#3,#4);
    \node[below] at (#1,#2) {Red = 0};
    \node[below] at (#1+#3,#2) {Blue = 1};
  \end{scope}
}
