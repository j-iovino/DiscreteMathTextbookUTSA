\section{Topological Spaces}
\label{sec:topology}

\begin{definition}
  \label{def:topological-space}
  A \emph{topological space} is a structure $(X, \scrN_x)_{x\in X}$  that consists of a set $X$ and, for each point $x\in X$, a collection $\scrN_x$ of subsets of $X$ that satisfy the following properties:

  \begin{enumerate}
  \item
    $x\in U$ for every $U\in \scrN_x$;
  \item
    If $U,V\in \scrN_x$, then there exists $W\in \scrN_x$ such that $W\subseteq U\cap V$;
  \item
    For every $U\in \scrN_x$ and every $y\in U$, there exists $V_y\in \scrN_y$ such that $V_y\subseteq U$.

  \end{enumerate}
\end{definition}

Next we introduce an immediate result from the previous definition.
The reader should note that Property (3) and the next proposition are equivalent.

\begin{proposition}
  \label{prop:basic-nhoods-are-open}
  Let $(X, \scrN_x)_{x\in X}$ be a topological space.
  For every $U\in \scrN_x$ and every $y\in U$, there exists $V_y\in \scrN_y$ such that \[U=\bigcup\limits_{y\in U}V_y\text{.}\]
\end{proposition}

\begin{proof}
  Fix a point $x\in X$, and let $U\in \scrN_x$ be given.
  By Property (3) above, we have that for every $y\in U$, there is some $V_y\in \scrN_y$ such that $V_y\subseteq U$.
  Thus, the union of all such $V_y$ is contained in $U$.
  Also, by Property (1), we see that for every $y\in U$, we have $y\in V_y\subseteq \bigcup\limits_{y\in U}V_y$.
  Thus, $U=\bigcup\limits_{y\in U}V_y$.
\end{proof}

\begin{definition}
  Let $( X, \scrN_x)_{x\in X}$ be a topological space.

  For each $x\in X$, we refer to $\scrN_x$ as the \emph{neighborhood base} for $x$.
  The indexed family $(\scrN_x)_{x\in X}$ is called a \emph{local neighborhood base} on $X$.
  For all $x\in X$, each set $U\in \scrN_x$ is called a \emph{basic neighborhood} of $x$.
  A \emph{neighborhood} of $x$ is any superset of a basic neighborhood of $x$.

\end{definition}

\begin{definition}
  A set $O\subseteq X$ is called \emph{open} if for every $x\in O$, there exists $V_x\in \scrN_x$ such that
  \[
    O=\bigcup\limits_{x\in O}V_x\text{.}
  \]

  A set $C\subseteq X$ is called \emph{closed} if $X\setminus C$ is open.
  Note that $C$ is closed if for every $x\in X$, and every basic neighborhood of $x$ that intersects $C$, then $x\in C$.
\end{definition}

Note that, by \cref{prop:basic-nhoods-are-open}, a subset of $X$ is open if and only it is a union of basic neighborhoods.

\begin{proposition}
  \label{prop:properties-of-open-sets}

  In any topological space $(X, \scrN_x)_{x\in X}$, the following properties hold:

  \begin{enumerate}
  \item
    Both $\emptyset$ and $X$ are open subsets of $X$ (and evidently also closed);
  \item
    Any union of open subsets of $X$ is open;
  \item
    Any finite intersection of open subsets of $X$ is open.

  \end{enumerate}
\end{proposition}

\begin{proof}

  \leavevmode

  \begin{enumerate}
  \item
    The empty set is open since any condition that must be satisfied by the points in $\emptyset$ is true vacuously.
    The set $X$ itself is open since, for each point $x\in X$, we may choose $V_x\in \scrN_x$ arbitrarily so that $X=\bigcup\limits_{x\in X}V_x$.

  \item
    Given any open subsets $(O_i)_{i\in I}$ of $X$, we have that each $O_i$ is a union of basic neighborhoods.
    Thus, the union $\bigcup\limits_{i\in I}O_i$ is a larger union of basic neighborhoods, and hence also an open subset of $X$.

  \item
    If $O_1$ and $O_2$ are open subsets of $X$, then for each $x\in O_1\cap O_2$, there are $U,V\in \scrN_x$ such that $U\subseteq O_1$ and $V\subseteq O_2$.
    By Property (2) of \cref{def:topological-space}, there’s
    $W_x\in \scrN_x$ such that \[W_x\subseteq U\cap V\subseteq O_1\cap O_2\text{.}\] Thus, $O_1\cap O_2$ is the union of all such $W_x\in \scrN_x$ that satisfy $W_x\subseteq O_1\cap O_2$ as $x$ ranges over $O_1\cap O_2$.
  \end{enumerate}
\end{proof}

Note that an infinite intersection of open sets is not necessarily open, as will be shown in \cref{exa:basic-topological-spaces}(a).

\begin{definition}
  Let $(X, \scrN_x)_{x\in X}$ be a topological space.

  The \emph{topology} $\scrT$ on $X$ generated by the local neighborhood base $(\scrN_x)_{x\in X}$  is the collection of all open subsets of $X$;
  that is,
  \[
    \scrT=
    \{ O\subseteq X \colon O \text{ is open}\}\text{.}
  \]
\end{definition}

Note that two different local neighborhood bases on $X$ can generate the same topology.
For instance, examples (a) and (b) below generate the same topology on $\RR$.

\begin{example} Basic examples
  \label{exa:basic-topological-spaces}

  \begin{enumerate}[(a)]

  \item
    Let $X=\RR$, and for each $x\in \RR$,  let $\scrN_x=\{(x-\varepsilon , x+\varepsilon):\varepsilon >0\}$.

    We will show that there is an infinite intersection of open subsets of $X$ that is not open.
    Take, for instance, the infinite intersection \[\bigcap\limits_{n\in \NN}(0,1+\frac{1}{n})=(0,1]\text{.}\]
    This interval is not open since there is no basic neighborhood $(1-\varepsilon,1+\varepsilon)$ of $1\in X$ that is contained in $(0,1]$.

  \item
    Let $X=\RR$, and for each $x\in \RR$,  let $\scrN_x=\{(\alpha , \beta) \colon \alpha<x<\beta, \alpha,\beta\in\RR \}$.

  \item
    Let $X=\RR^2$, and for each $x\in \RR^2$, let $\scrN_x$ consist of the set of all open disks about $x$ of radius $\varepsilon$ for all $\varepsilon >0$.

  \item
    Let $X$ be any set, and for each $x\in X$, let $\scrN_x = \{\{x\}\}$;
    i.e., the only basic neighborhood of $x$ is the singleton $\{x\}$.
    The resulting topology is called the \emph{discrete} topology on $X$.

    Note that, in this topological space, every subset of $X$ is open (since every set is a union of singletons).
    Hence, every subset is closed as well.

  \item
    Let $X$ be any set, and for each $x\in X$, let $\scrN_x = \{X\}$.
    The resulting topology is called the \emph{trivial} (or \emph{indiscrete}) topology on $X$.
    Note that, in this topological space, only two subsets are open;
    namely, $\emptyset$ and $X$.

  \end{enumerate}

\end{example}

  \end{enumerate}

\section{Topological Limits}

We now transition to discuss the notion of limit in a topological space.
Fix a topological space $(X,\scrN_x)_{x\in X}$ and an indexed family $(a_i)_{i\in I}$ in $X$.
We want to express what it means for $(a_i)_{i\in I}$ to converge to a point $c\in X$.
Intuitively, this means that $(a_i)_{i\in I}$ \emph{clusters} around a point $c\in X$;
that is, for every basic neighborhood $V_c\in \scrN_c$, there exists a “large” $L\subseteq I$ such that $\{a_i \colon i\in L\}\subseteq V_c$.
To clarify what is meant by “large,” we proceed by defining the notion of filter, which may be thought of as a collection of all the “large” subsets $L$ of $I$.

\begin{definition}
  \label{def:filter}

  Fix a nonempty set $I$.
  A \emph{filter} on $I$ is a collection $\scrF$ of subsets of $I$ that satisfy the following properties:

  \begin{enumerate}
  \item
    $\emptyset\not\in\scrF$ and $I\in \scrF$;
  \item
    If $A\in\scrF$ and $B\supseteq A$, then $B\in\scrF$;
  \item
    If $A,B\in \scrF$, then $A\cap B\in\scrF$.

  \end{enumerate}
\end{definition}

Similarly, we may define the notion of “small” in the following way.

\begin{definition}
  \label{def:ideal}

  Fix a nonempty set $I$.
  An \emph{ideal} on $I$ is a collection $\scrJ$ of subsets of $I$ that satisfy the following properties:

  \begin{enumerate}
  \item
    $\emptyset\in\scrJ$ and $I\not\in \scrJ$;
  \item
    If $A\in\scrJ$ and $B\subseteq A$, then $B\in\scrJ$;
  \item
    If $A,B\in \scrJ$, then $A\cup B\in\scrJ$.

  \end{enumerate}

\end{definition}

Observe that given a filter $\scrF$ on a nonempty set $I$, we can define an ideal $\scrJ$ on $I$ by \[\scrJ=\{I\setminus A \colon A\in\scrF\}\text{.}\]
Similarly, given an ideal $\scrJ$ on $I$, we see that \[\scrF = \{I\setminus A \colon A\in \scrJ\}\] is a filter on $I$.

\begin{notation}
  \label{notn:large-sets}
  If $\scrF$ is a filter on a nonempty set $I$, then the elements of $\scrF$ are called \emph{$\scrF$-large}.
  A property of elements of $I$ is said to hold for \emph{almost every $i\in I$} or \emph{almost everywhere} in $I \pmod{\scrF}$ if the property holds for an $\scrF$-large subset of $I$.
\end{notation}

We are now ready to define the notion of convergence.

\begin{definition}
  Let $(X, \scrN_x)_{x\in X}$ be a topological space, and let $(a_i)_{i\in I}$ be an indexed family of elements in $X$.
  If $\scrF$ is a filter on $I$, we say $a_i$ \emph{converges} to a point $c\in X$ with respect to $\scrF$, denoted $a_i\overset{\scrF}\to c$, if and only if for every $V_c\in\scrN_c$, we have $\{i \colon a_i\in V_c\}\in\scrF$.

  We call $c$ an \emph{$\scrF$-limit} of $(a_i)_{i\in I}$ if and only if $(a_i)_{i\in I}$ converges to $c$ with respect to $\scrF$.

\end{definition}

In the context of a sequence of real numbers, this is the definition we are familiar with.
Consider \cref{exa:basic-topological-spaces}(a);
namely, the topological space $(\RR,\scrN_x)_{x\in \RR}$ where $\scrN_x = \{(x-\varepsilon,x+\varepsilon) \colon \varepsilon>0\}$.
Let $I=\NN$, and let $\scrF$ consist of all cofinite subsets of $\NN$.
Then, if $(a_n)_{n\in \NN}$ is a sequence of real numbers, we see that $a_n \overset{\scrF}\to c$ if and only if for every $V_c\in\scrN_c$, we have $a_n\in V_c$ for all but finitely many $n\in\NN$;
that is, for all $\varepsilon >0$, there exists  $N\in\NN$ such that for all $n>N$, we have $a_n\in(c-\varepsilon,c+\varepsilon)$.

\begin{definition}
  The filter that consists of all cofinite subsets of any nonempty set $I$ is called the \emph{Fréchet filter on $I$}.
\end{definition}

Recall that, in the topological space $(\RR,\scrN_x)_{x\in \RR}$ mentioned above, limits are unique.
However, this is not true in every topological space.
Consider the indiscrete topology discussed in \cref{exa:basic-topological-spaces}(e);
namely, given any set $X$, the indiscrete topology on $X$ is the set $\scrT=\{\emptyset, X\}$.
Given any indexed family $(a_i)_{i\in I}$ in $X$ and any filter $\scrF$ on $I$, since $X$ is the only basic neighborhood of any point in $X$ and \[\{i\colon a_i\in X\}=I\in\scrF\text{,}\] it follows that every point in $X$ is a limit of $(a_i)_{i\in I}$.

Thus, a new question arises:
What property must a topological space have to ensure that every limit is unique?
This question leads us to the following definition.

\begin{definition}
  A topological space $(X,\scrN_x)_{x\in X}$ is called \emph{Hausdorff} if any two distinct points in $X$ can be separated by disjoint neighborhoods;
  i.e., if $x,y\in X$ such that $x\not =y$, then there exist basic neighborhoods $V_x\in\scrN_x$ and $V_y\in\scrN_y$ such that $V_x\cap V_y=\emptyset$.
\end{definition}

\begin{proposition}
  \label{prop:hausdorff-implies-unique-limits}
  Let $(X,\scrN_x)_{x\in X}$ be a Hausdorff topological space, let $(a_i)_{i\in I}$ be an indexed family in $X$, and let $\scrF$ be a filter on $I$.
  If $a_i\overset{\scrF}\to c$ and $a_i\overset{\scrF}\to d$, then $c=d$.
\end{proposition}

\begin{proof}
  Suppose, on the contrary, that
  \begin{align*}
  a_i\overset{\scrF}\to c \text{,} \\
  a_i\overset{\scrF}\to d\text{,}
  \end{align*}
  and $c\not=d$.
  Since $(X,\scrN_x)_{x\in X}$ is Hausdorff, we can separate $c$ and $d$ by disjoint neighborhoods, say, $U$ and $V$.
  Then we have
  \begin{align*}
  A & =\{i \colon a_i\in U\}\in\scrF \text{ and} \\
  B & =\{i \colon a_i\in V\}\in\scrF\text{,}
  \end{align*}
  so by Property (3) of \cref{def:filter}, we have $A\cap B=\emptyset\in \scrF$, which contradicts Property (1).
  We conclude $c=d$.
\end{proof}

The converse of \cref{prop:hausdorff-implies-unique-limits} is also true.
Although it is not needed for the material in these notes, it can be proved easily, so we have included a proof in Proposition~\ref{P:unique limits->Hausdorff}

\begin{notation}
  In case $\scrF$-limits are unique in a topological space $(X,\scrN_x)_{x\in X}$ and an indexed family $(a_i)_{i\in I}$ in $X$ converges to $c\in X$, we write
  \[
    c=\scrF\lim a_i \quad \text{or} \quad c= \lim\limits_{i,\scrF}a_i \text{.}
  \]
\end{notation}

\subsection{Principal Ultrafilters \nopunct}

\begin{definition}
  An \emph{ultrafilter} $\scrU$ on a nonempty set $I$ is a filter on $I$ such that for every $A\subseteq I$, either $A\in \scrU$ or $I\setminus A\in \scrU$.
\end{definition}

Filters can also be seen as generalized measures.
An ultrafilter $\scrU$ on a set $I$ divides the subsets of $I$ into two categories:
$\scrU$-large and $\scrU$-small --- the large subsets of $I$ are those in $\scrU$, and the small subsets of $I$ are those whose complement is in $\scrU$.
Thus, ultrafilters are “complete” measures in the sense that every set is measurable.
Filters, in contrast, need not be complete in this sense:
For an arbitrary filter $\scrF$ on a set $I$, there may be subsets of
$I$ that are neither $\scrF$-large nor $\scrF$-small (for instance, for the Fréchet filter on $\NN$, any subset of $\NN$ that is neither finite nor cofinite --- e.g., the set of even numbers --- is neither $\scrF$-large nor $\scrF$-small.)

\begin{example} Principal Ultrafilters

  Fix a nonempty set $I$ and an element $a\in I$.
  Define
  \[
    \scrU_a=\{A\subseteq I \colon a\in A\}.
  \]
  Note that $\scrU_a$ is an ultrafilter on the set $I$.
  Any ultrafilter defined in this way is called a \emph{principal} (or “trivial” or “fixed”) ultrafilter.
  More precisely, $\scrU_a$ is called the principal ultrafilter on $I$ \emph{generated by $a$}.
\end{example}

The preceding example suggests that ultrafilters represent an abstract notion of “point." Principal ultrafilters are “hard" points, while nonprincipal ultrafilters are “soft" points.

\begin{exercise}
  Prove that if $I$ is finite, then every ultrafiter on $I$ is principal.
\end{exercise}

\subsection{Nonprincipal Ultrafilters \nopunct}

\begin{proposition}
  An ultrafilter $\scrU$ on a set $I$ is principal if and only if $\scrU$ contains a finite subset of $I$.
\end{proposition}

\begin{proof}
  If $\scrU$ is principal, then $\scrU$ contains a finite set;
  namely, the singleton that generates it.
  Suppose, conversely, that $\scrU$ is nonprincipal.
  We prove, by induction on $n$, that $\scrU$ cannot contain $n$-element subsets of $I$ for $n=1, 2,\dots$.
  The case $n=1$ is given by the definition of nonprincipal.
  Suppose that $\scrU$ contains a set $\{a_1,\dots, a_n\}$, where $a_1,\dots, a_n$ are distinct elements of $I$.
  Since, by the non nonprincipality assumption, we cannot have $\{a_1\}\in \scrU$, we must have $I\setminus \{a_1\}\in \scrU$.
  But then, since $\scrU$ is closed under finite intersections, $\scrU$ must contain $(I\setminus \{a_1\})\cap\{a_1,\dots, a_n\}=\{a_2,\dots, a_n\}$, which contradicts the induction assumption.
\end{proof}

The preceding proposition can be re-stated as follows:

\begin{proposition}
  \label{prop:nonprincipality-and-frechet}
  An ultrafilter $\scrU$ on an infinite set $I$ is nonprincipal if and only if it extends the Fréchet filter on $I$.
\end{proposition}

In light of the preceding proposition, to prove that nonprincipal ultrafilters exist, we only have to prove that the Fréchet filter on $I$ can be extended to an ultrafilter.
The following propositions show that, indeed,  any collection of subsets of $I$ that has the finite intersection property can be extended to an ultrafilter;
see, in particular, \cref{cor:ultrafilter-theorem,cor:nonprincipal-ultrafilters-exist}.

\begin{definition}
  \label{def:fip}
  A collection $\scrB$ of subsets of $I$ has the \emph{finite intersection property} (or is \emph{finitely consistent}) if whenever $A_1,\dots, A_n\in\scrB$, we have
  \[
    A_1\cap\dots\cap A_n\neq \emptyset \text{.}
  \]
\end{definition}

The finite intersection property is ubiquitous in mathematics.
Mathematicians often refer to it as “f.i.p."

\begin{proposition}
  \label{prop:filter-generated-by-a-base}
  Every collection of subsets of $I$ that has the finite intersection property can be extended to a filter on $I$.
\end{proposition}

\begin{proof}
  Let $\scrB$ be a collection of subsets of $I$ with the finite intersection property.
  Without loss of generality, we can assume that $\scrB$ is closed under finite intersections.
  But then, once we have assumed this, to extend $\scrB$ to a filter, we only have to close it under supersets;
  in other words, the collection
  \[
    \scrF=\{A\subseteq I \colon A\supseteq B
    \text{ for some $B\in \scrB$} \}
  \]
  is clearly a filter on $I$.
\end{proof}

Due to the preceding proposition, any collection $\scrB$ of subsets of $I$ with the finite intersection property is often called a \emph{base} for a filter (or a \emph{filter base}) on $I$.
The filter $\scrF$ constructed in the proof of the \cref{prop:filter-generated-by-a-base} is called the filter \emph{generated} by $\scrB$.

Now we want to improve \cref{prop:filter-generated-by-a-base} by showing that every filter can be extended to an ultrafilter on $I$.

\begin{proposition}
  \label{prop:ultrafilter-lemma}
  Let $\scrF$ be a filter on a set $I$.
  Then,  for every $A\subseteq I$, there exists a filter $\scrF'\supseteq \scrF$ on $I$ such that either $A\in \scrF'$ or $I\setminus A \in \scrF'$
\end{proposition}

\begin{proof}
  Fix a set $A\subseteq I$.
  By \cref{prop:filter-generated-by-a-base}, it suffices to verify that either $\scrF\cup\{A\}$ or $\scrF\cup\{I\setminus A\}$ has the finite intersection property.
  But this is immediate because if $\scrF\cup\{A\}$ fails to have the finite intersection property, it is because there exist $A_1,\dots, A_n\in \scrF$ such that
  $A_1 \cap\dots \cap A_n\cap A=\emptyset$, and this means that
  $I\setminus A\supseteq A_1 \cap\dots \cap A_n$,
  which implies that $\scrF\cup\{I\setminus A\}$ has the finite intersection property.
\end{proof}

\begin{corollary}[{The Ultrafilter Theorem [A. Tarski, 1930]}]
  \label{cor:ultrafilter-theorem}
  Every filter on a set $I$ can be extended to an ultrafilter on $I$.
\end{corollary}

\begin{proof}
  Fix a filter $\scrF$ on $I$,  let $\alpha=|\scrP(I)|$ and let $(A_\xi)_{\xi<\alpha}$ be a list of all the subsets of $I$.
  By induction, we construct an increasing chain
  \[
    \scrF=\scrF_0\subseteq \scrF_1\subseteq \dots\subseteq \scrF_\xi\subseteq \dots\qquad (\xi\le\alpha)
  \]
  of filters in $I$  such that:
  \begin{enumerate}
  \item
    If $\xi = \zeta +1$, then
    \[
      A_\zeta\in \scrF_\xi
      \quad\text{or}\quad
      I\setminus A_\zeta\in \scrF_\xi;
    \]
  \item
    If $\xi\le\alpha$ is a nonzero limit ordinal, then
    \[
      \scrF_\xi = \bigcup_{\zeta <\xi} \scrF_\zeta\text{.}
    \]
  \end{enumerate}

  The construction is as follows.
  Fix $\xi<\alpha$ and that $\scrF_\zeta$ has been defined for all $\zeta<\xi$.

  If $\xi$ is a successor ordinal, say $\xi=\zeta+1$, by \cref{prop:ultrafilter-lemma},
  either $\scrF_\zeta \cup \{A_\zeta\}$ or $\scrF_\zeta \cup \{I\setminus A_\zeta\}$ has the finite intersection property (it is possible that both sets have the f.i.p.).
  We pick one of these sets that has the finite intersection property, and let $\scrF_\xi$ be a filter extending it.

  If $\xi$ is a limit ordinal, we simply let $\scrF_\xi = \bigcup_{\zeta<\xi}\scrF_\zeta$.
  Since $(\scrF_\zeta)_{\zeta<\xi}$ is an increasing chain of filters, it is easy to see that
  $\scrF_\xi$ is closed under finite intersections and supersets (for if $A,B\in \scrF_\xi$, then $A$ and $B$ are in some member of the chain), and does not contain the empty set, i.e., $\scrF_\xi$ is a filter.

  By construction, $\scrF_\alpha$ is an ultrafilter containing $\scrF_0=\scrF$.

\end{proof}

\begin{corollary}
  \label{cor:nonprincipal-ultrafilters-exist}
  Nonprincipal ultrafilters exist on any infinite set.
\end{corollary}

\begin{proof}
  Let $I$ be an infinite set.
  By \cref{prop:ultrafilter-lemma}, the Fréchet filter on $I$ can be extended to an ultrafilter.
  By \cref{prop:nonprincipality-and-frechet}, such an ultrafilter must be nonprincipal.
\end{proof}

\begin{corollary}
  \label{cor:base-extended-to-ultrafilter}
  Let $\scrB$ be a family of subsets of $I$ with the finite intersection property.
  Then there exists an ultrafilter $\scrU$ on $I$ such that every set in $\scrB$ is $\scrU$-large.
\end{corollary}

\begin{proof}
  By \cref{prop:filter-generated-by-a-base} and \cref{cor:ultrafilter-theorem}.
\end{proof}
